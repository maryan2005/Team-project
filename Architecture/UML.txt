' Main Engine Controller
class GameEngine {
    - screen : pygame.Surface
    - clock : pygame.time.Clock
    - is_running : bool
    - target_fps : int
    - current_state : GameState
    + settings : Dict[str, Any]
    
    + __init__(width: int, height: int, title: str, target_fps: int = 60)
    + initialize() : void
    + run() : void
    + change_state(new_state: GameState) : void
    + handle_events() : void
    + update(delta_time: float) : void
    + render() : void
    + quit() : void
}

' Abstract Base State
abstract class GameState {
    # engine : GameEngine
    + __init__(engine: GameEngine)
    + {abstract} handle_input(event: pygame.event.Event) : void
    + {abstract} update(delta_time: float) : void
    + {abstract} draw(surface: pygame.Surface) : void
}

' Engine & State Relationships
GameEngine "1" *-- "1" GameState : current_state
GameState "1" o-- "1" GameEngine : engine

' Inheritance Hierarchy
GameState <|-- TitleState
GameState <|-- PlayState
GameState <|-- PauseState
GameState <|-- SettingsState
GameState <|-- GameOverState

' Specific Pause State Relationship
PauseState "1" o-- "1" PlayState : play_state

' PlayState Domain Aggregations & Compositions
PlayState "1" *-- "1" InputHandler : input_handler
PlayState "1" *-- "1" Board : board
PlayState "1" *-- "1" ScoreManager : score_manager
PlayState "1" *-- "1" ChallengeManager : challenge_manager
PlayState "1" o-- "1..3" Piece : active, next, held

' Board and Piece Interaction
Board "1" ..> "1" Piece : validates/locks

' PlayState Interaction with Managers
PlayState "1" ..> "1" ScoreManager : updates on action
PlayState "1" ..> "1" ChallengeManager : evaluates progress
